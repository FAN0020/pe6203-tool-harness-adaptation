## The core principle: replace `eval()` with a whitelist grammar

The single most important rule for secure math in Python: **never pass untrusted input to `eval()`, `exec()`, or `compile()`.** Instead, parse into an AST and evaluate **only an explicitly allowed set of nodes/operators** — a whitelist grammar.

The `validate_syntax` / `evaluate` tools you have are a working example of exactly this. Their behavior on my test inputs:

| Input | Result | Why it matters |
|---|---|---|
| `2 + 3 * (4 - 1) / 2` | ✅ **valid → 6.5** | Legitimate arithmetic works |
| `10 % 3` / `-5 + 3` | ✅ `1` / `-2` | Operators & unary handled |
| `__import__('os').system('rm -rf /')` | ❌ rejected (`Unexpected character '_'`) | **RCE attempt blocked** (A03) |
| `open('/etc/passwd').read()` | ❌ rejected | Attribute/IO access blocked |
| `math.sqrt(16)` | ❌ rejected | Function calls blocked |
| `9**9**9**9` | ❌ rejected | **Exponent-bomb DoS blocked** (A04) |
| `10 / 0` | ⚠️ `null` (no crash) | Graceful, sanitized failure |
| `1e309` | ⚠️ `null` | Overflow handled, no traceback leak |

Two lessons from this: (1) the whitelist blocks injection *and* function/attribute access; (2) errors fail **closed and quietly** — no stack traces leaked back to the caller.

## Reference implementation: a safe AST evaluator

```python
import ast
import operator

# ---- Limits (tune to your workload) ----
MAX_INPUT_LEN = 1_000      # total characters
MAX_NODES     = 200        # AST size / nesting cap  -> blocks deep recursion & giant trees
MAX_INT_BITS  = 4_096      # max size of any integer  -> blocks huge-int DoS
MAX_EXPONENT  = 1_000      # cap on `a ** b`          -> blocks exponent bombs

_BIN_OPS = {
    ast.Add: operator.add,        ast.Sub: operator.sub,
    ast.Mult: operator.mul,       ast.Div: operator.truediv,
    ast.FloorDiv: operator.floordiv, ast.Mod: operator.mod,
    ast.Pow: operator.pow,
}
_UNARY_OPS = {ast.UAdd: operator.pos, ast.USub: operator.neg}


class SafeEvalError(ValueError):
    """Raised on any disallowed or unsafe expression."""


def safe_eval(expr: str) -> int | float:
    if not isinstance(expr, str):
        raise SafeEvalError("expression must be a string")
    if len(expr) > MAX_INPUT_LEN:
        raise SafeEvalError("expression too long")

    try:
        tree = ast.parse(expr, mode="eval")
    except SyntaxError as exc:
        raise SafeEvalError("invalid syntax") from exc

    if sum(1 for _ in ast.walk(tree)) > MAX_NODES:
        raise SafeEvalError("expression too large or deeply nested")

    return _eval(tree.body)


def _eval(node):
    # 1) Numeric literals only (rejects str, bytes, bool, complex)
    if isinstance(node, ast.Constant):
        if isinstance(node.value, bool) or not isinstance(node.value, (int, float)):
            raise SafeEvalError("only numeric literals are allowed")
        return node.value

    # 2) Binary operators — whitelist only
    if isinstance(node, ast.BinOp):
        op = _BIN_OPS.get(type(node.op))
        if op is None:
            raise SafeEvalError(f"operator not allowed: {type(node.op).__name__}")
        left, right = _eval(node.left), _eval(node.right)

        if isinstance(node.op, (ast.Div, ast.FloorDiv, ast.Mod)) and right == 0:
            raise SafeEvalError("division by zero")
        if isinstance(node.op, ast.Pow) and abs(right) > MAX_EXPONENT:
            raise SafeEvalError("exponent too large")

        result = op(left, right)
        if isinstance(result, int) and result.bit_length() > MAX_INT_BITS:
            raise SafeEvalError("result too large")
        return result

    # 3) Unary + / -
    if isinstance(node, ast.UnaryOp):
        fn = _UNARY_OPS.get(type(node.op))
        if fn is None:
            raise SafeEvalError("unary operator not allowed")
        return fn(_eval(node.operand))

    # 4) Everything else (Call, Attribute, Name, Lambda, ListComp, ...) is rejected
    raise SafeEvalError(f"disallowed expression: {type(node).__name__}")
```

Key safety properties: no `Name`/`Attribute`/`Call` nodes are ever evaluated, so there is **no path to imports, I/O, or attribute access**.

### ✅ Safe: AST whitelist evaluator
```python
import ast
import operator

# Only these node types are allowed
_ALLOWED_BINOPS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.FloorDiv: operator.floordiv,
    ast.Mod: operator.mod,
    ast.Pow: operator.pow,
}
_ALLOWED_UNARY = {ast.UAdd: operator.pos, ast.USub: operator.neg}
_ALLOWED_FUNCS = {"abs": abs, "round": round, "min": min, "max": max}

MAX_EXPR_LEN = 256
MAX_POW_EXPONENT = 1000
MAX_DEPTH = 20

class UnsafeExpression(ValueError):
    pass

def _eval_node(node, depth=0):
    if depth > MAX_DEPTH:
        raise UnsafeExpression("Expression too deeply nested")

    if isinstance(node, ast.Expression):
        return _eval_node(node.body, depth + 1)

    if isinstance(node, ast.Constant):
        if isinstance(node.value, (int, float)) and not isinstance(node.value, bool):
            return node.value
        raise UnsafeExpression("Only numeric constants allowed")

    if isinstance(node, ast.BinOp):
        op = _ALLOWED_BINOPS.get(type(node.op))
        if op is None:
            raise UnsafeExpression(f"Operator not allowed: {type(node.op).__name__}")
        left = _eval_node(node.left, depth + 1)
        right = _eval_node(node.right, depth + 1)
        # Guard against exponent bombs
        if isinstance(node.op, ast.Pow) and abs(right) > MAX_POW_EXPONENT:
            raise UnsafeExpression("Exponent too large")
        return op(left, right)

    if isinstance(node, ast.UnaryOp):
        op = _ALLOWED_UNARY.get(type(node.op))
        if op is None:
            raise UnsafeExpression("Unary operator not allowed")
        return op(_eval_node(node.operand, depth + 1))

    if isinstance(node, ast.Call):
        if not isinstance(node.func, ast.Name) or node.func.id not in _ALLOWED_FUNCS:
            raise UnsafeExpression("Function not allowed")
        if node.keywords:
            raise UnsafeExpression("Keyword arguments not allowed")
        args = [_eval_node(a, depth + 1) for a in node.args]
        return _ALLOWED_FUNCS[node.func.id](*args)

    raise UnsafeExpression(f"Disallowed syntax: {type(node).__name__}")

def safe_eval(expr: str) -> float:
    if not isinstance(expr, str):
        raise UnsafeExpression("Expression must be a string")
    if len(expr) > MAX_EXPR_LEN:
        raise UnsafeExpression("Expression too long")
    try:
        tree = ast.parse(expr, mode="eval")
    except SyntaxError as e:
        raise UnsafeExpression(f"Invalid syntax: {e.msg}") from e
    return _eval_node(tree)
```

This mirrors exactly what the validation tool did: parse → reject anything outside the grammar → evaluate only whitelisted operations.

```python
import ast
import operator
import logging

logger = logging.getLogger("safe_math")

# Layer 1: allowlist of characters + length cap
ALLOWED_CHARS = set("0123456789+-*/(). ")
MAX_LEN = 200

# Layer 2: only these AST nodes are permitted
_ALLOWED_NODES = (
    ast.Expression, ast.BinOp, ast.UnaryOp, ast.Constant,
    ast.Add, ast.Sub, ast.Mult, ast.Div, ast.Pow, ast.Mod,
    ast.USub, ast.UAdd, ast.FloorDiv,
)

    ast.Add: operator.add, ast.Sub: operator.sub,
    ast.Mult: operator.mul, ast.Div: operator.truediv,
    ast.Pow: operator.pow, ast.Mod: operator.mod,
    ast.FloorDiv: operator.floordiv,
}

    pass

    if not isinstance(expr, str):
        raise UnsafeExpression("Expression must be a string")

        raise UnsafeExpression("Illegal characters in expression")

    if isinstance(node, ast.Expression):
        return _eval_node(node.body)
    if isinstance(node, ast.Constant):
        if isinstance(node.value, (int, float)) and not isinstance(node.value, bool):
            return node.value

    if isinstance(node, ast.UnaryOp) and type(node.op) in _UNARY_OPS:

        tree = ast.parse(expr, mode="eval")
```

```python
import ast
import operator
from decimal import Decimal, DivisionByZero, InvalidOperation

MAX_LEN, MAX_DEPTH = 256, 12
MAX_ABS_EXPONENT = 64

class ExpressionRejected(ValueError):
    """Message text is fixed and safe to return to clients."""

_BINOPS = {ast.Add: operator.add, ast.Sub: operator.sub,
           ast.Mult: operator.mul, ast.Div: operator.truediv,
           ast.Mod: operator.mod}
_UNARYOPS = {ast.UAdd: operator.pos, ast.USub: operator.neg}

def _check_depth(node: ast.AST) -> None:
    # Iterative on purpose: the checker must not hit the recursion limit itself.
    stack = [(node, 1)]
    while stack:
        cur, d = stack.pop()
        if d > MAX_DEPTH:
            raise ExpressionRejected("expression is too deeply nested")
        stack.extend((c, d + 1) for c in ast.iter_child_nodes(cur))

def safe_eval(expr: str, variables: dict[str, Decimal] | None = None) -> Decimal:
    if not isinstance(expr, str) or len(expr) > MAX_LEN:
        raise ExpressionRejected("expression is invalid")
    variables = variables or {}
    try:
        tree = ast.parse(expr, mode="eval")   # mode="eval" blocks statements
    except SyntaxError:
        raise ExpressionRejected("expression is invalid") from None
    _check_depth(tree)
    return _eval(tree.body, variables)

def _eval(node: ast.AST, variables: dict[str, Decimal]) -> Decimal:
    if isinstance(node, ast.Constant):
        # bool is an int subclass — reject it explicitly
        if isinstance(node.value, bool) or not isinstance(node.value, (int, float)):
            raise ExpressionRejected("unsupported literal")
        return Decimal(str(node.value))
    if isinstance(node, ast.Name):
        if node.id not in variables:          # allowlist lookup, never globals()
            raise ExpressionRejected("unknown variable")
        return Decimal(variables[node.id])
    if isinstance(node, ast.BinOp) and type(node.op) in _BINOPS:
        return _BINOPS[type(node.op)](_eval(node.left, variables),
                                      _eval(node.right, variables))
    if isinstance(node, ast.UnaryOp) and type(node.op) in _UNARYOPS:
        return _UNARYOPS[type(node.op)](_eval(node.operand, variables))
    raise ExpressionRejected("unsupported construct")
```

## 2. The Secure Pattern: Whitelist via AST, Never `eval`

```python
import ast

# 1. Whitelist of allowed binary/unary operators
_ALLOWED_BINOPS = {
    ast.Add: op.add, ast.Sub: op.sub, ast.Mult: op.mul,
    ast.Div: op.truediv, ast.FloorDiv: op.floordiv,
    ast.Mod: op.mod, ast.Pow: op.pow,
}
_ALLOWED_UNARYOPS = {ast.UAdd: op.pos, ast.USub: op.neg}

# 2. Whitelist of allowed functions (NO builtins, NO imports)
_ALLOWED_FUNCS = {
    "sqrt": math.sqrt, "abs": abs, "min": min, "max": max,
    "round": round, "log": math.log, "exp": math.exp,
}

class UnsafeExpressionError(ValueError):
    pass

def safe_eval(expr: str, max_len: int = 200, max_depth: int = 20) -> float:
    # --- Guard rails BEFORE parsing (DoS protection, A04) ---
    if not isinstance(expr, str) or len(expr) > max_len:
        raise UnsafeExpressionError("Expression too long or wrong type")

    try:
        tree = ast.parse(expr, mode="eval")
    except SyntaxError as e:
        raise UnsafeExpressionError(f"Invalid syntax: {e}") from e

    # --- Depth guard (prevents deeply nested / exponential blowups) ---
    if _depth(tree) > max_depth:
        raise UnsafeExpressionError("Expression nesting too deep")

    return _eval_node(tree.body)

def _depth(node, level=1):
    children = list(ast.iter_child_nodes(node))
    return level if not children else max(_depth(c, level + 1) for c in children)

def _eval_node(node):
    if isinstance(node, ast.Constant):                 # numbers only
        if isinstance(node.value, (int, float)) and not isinstance(node.value, bool):
            return node.value
        raise UnsafeExpressionError("Only numeric constants allowed")
    if isinstance(node, ast.BinOp) and type(node.op) in _ALLOWED_BINOPS:
        return _ALLOWED_BINOPS[type(node.op)](_eval_node(node.left), _eval_node(node.right))
    if isinstance(node, ast.UnaryOp) and type(node.op) in _ALLOWED_UNARYOPS:
        return _ALLOWED_UNARYOPS[type(node.op)](_eval_node(node.operand))
    if isinstance(node, ast.Call):
        if not isinstance(node.func, ast.Name) or node.func.id not in _ALLOWED_FUNCS:
            raise UnsafeExpressionError(f"Function not allowed: {ast.dump(node.func)}")
        if node.keywords:
            raise UnsafeExpressionError("Keyword arguments not allowed")
        return _ALLOWED_FUNCS[node.func.id](*[_eval_node(a) for a in node.args])
    # Everything else (Attribute, Subscript, Name, Lambda, comprehensions...) is rejected
    raise UnsafeExpressionError(f"Disallowed construct: {type(node).__name__}")
```

## The only safe architecture: AST allowlist, never `eval`

Do not sanitize with regex, and do not call `eval`/`exec`/`compile` on user input — OWASP **A03: Injection**. Parse to an AST and walk only explicitly permitted node types.

```python
import ast
import math
import operator as op

MAX_LEN       = 256
MAX_DEPTH     = 12
MAX_EXPONENT  = 64
MAX_ABS_VALUE = 1e12

class UnsafeExpression(ValueError):
    """Raised for any rejected expression. Message is deliberately generic."""

_BINOPS = {
    ast.Add: op.add, ast.Sub: op.sub, ast.Mult: op.mul,
    ast.Div: op.truediv, ast.FloorDiv: op.floordiv,
    ast.Mod: op.mod, ast.Pow: op.pow,
}
_UNARY = {ast.UAdd: op.pos, ast.USub: op.neg}
_FUNCS = {
    "abs": abs, "min": min, "max": max, "round": round,
    "sqrt": math.sqrt, "floor": math.floor, "ceil": math.ceil,
}

def _depth(node: ast.AST, level: int = 0) -> int:
    children = list(ast.iter_child_nodes(node))
    return level if not children else max(_depth(c, level + 1) for c in children)

def safe_eval(expr: str, variables: dict[str, float] | None = None) -> float:
    variables = variables or {}
    if not isinstance(expr, str) or not expr or len(expr) > MAX_LEN:
        raise UnsafeExpression("expression rejected")

    try:
        tree = ast.parse(expr, mode="eval")
    except (SyntaxError, ValueError, MemoryError, RecursionError):
        raise UnsafeExpression("expression rejected") from None

    if _depth(tree) > MAX_DEPTH:
        raise UnsafeExpression("expression rejected")

    try:
        result = _eval(tree.body, variables)
    except UnsafeExpression:
        raise
    except (ArithmeticError, ValueError, TypeError, RecursionError):
        raise UnsafeExpression("expression rejected") from None

    if isinstance(result, complex) or not isinstance(result, (int, float)):
        raise UnsafeExpression("expression rejected")
    return float(result)

def _num(value: float) -> float:
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        raise UnsafeExpression("expression rejected")
    if abs(value) > MAX_ABS_VALUE:
        raise UnsafeExpression("expression rejected")
    return value

def _eval(node: ast.AST, variables: dict[str, float]) -> float:
    if isinstance(node, ast.Constant):          # rejects str/bytes/bool/None
        return _num(node.value)

    if isinstance(node, ast.Name):
        if node.id not in variables:
            raise UnsafeExpression("expression rejected")
        return _num(variables[node.id])

    if isinstance(node, ast.BinOp) and type(node.op) in _BINOPS:
        left, right = _eval(node.left, variables), _eval(node.right, variables)
        if isinstance(node.op, ast.Pow):
            if abs(right) > MAX_EXPONENT:       # guard BEFORE computing
                raise UnsafeExpression("expression rejected")
        return _num(_BINOPS[type(node.op)](left, right))

    if isinstance(node, ast.UnaryOp) and type(node.op) in _UNARY:
        return _num(_UNARY[type(node.op)](_eval(node.operand, variables)))

    if isinstance(node, ast.Call):
        if (not isinstance(node.func, ast.Name)
                or node.func.id not in _FUNCS
                or node.keywords
                or any(isinstance(a, ast.Starred) for a in node.args)
                or len(node.args) > 3):
            raise UnsafeExpression("expression rejected")
        args = [_eval(a, variables) for a in node.args]
        return _num(_FUNCS[node.func.id](*args))

    raise UnsafeExpression("expression rejected")   # Attribute, Subscript, Lambda, JoinedStr…
```

Why each guard matters:

- **`ast.Constant` type check** blocks the `"'; DROP TABLE users; --"` string that the syntax validator accepted.
- **`ast.Name` allowlist** makes `allow_variables` a real policy instead of a hint.
- **`_depth` on the AST** enforces nesting yourself, since the tool's `max_depth` did not.
- **Exponent guard before evaluation** stops `9**9**9` from hanging a worker — never compute first and check after.
- **`_num` result bound** catches overflow-to-infinity and `(-2) ** 0.5` producing `complex`.
- **Keyword/starred-arg rejection** closes `sqrt(x, **payload)` style tricks.
- **No `ast.Attribute` / `ast.Subscript` branch** is what kills `().__class__.__bases__[0].__subclasses__()` — the classic sandbox escape.

## 1. The Core Threat: Never `eval()` Untrusted Input

## 2. The Safe Pattern: AST Whitelist Evaluation

Parse the expression into an Abstract Syntax Tree and **allow only a fixed set of node types and operators**. This is the recommended defense-in-depth approach.

```python
import ast
import operator

# Whitelist of allowed binary/unary operators
_ALLOWED_OPERATORS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.FloorDiv: operator.floordiv,
    ast.Mod: operator.mod,
    ast.Pow: operator.pow,
    ast.USub: operator.neg,
    ast.UAdd: operator.pos,
}

MAX_EXPRESSION_LENGTH = 200
MAX_POWER_EXPONENT = 1000   # prevent 9**9**9 DoS

def safe_eval(expression: str) -> float:
    # 1. Input validation: length + type
    if not isinstance(expression, str) or len(expression) > MAX_EXPRESSION_LENGTH:
        raise ValueError("Expression too long or invalid type")

    # 2. Parse to AST (raises SyntaxError on malformed input)
    try:
        tree = ast.parse(expression, mode="eval")
    except SyntaxError as e:
        raise ValueError(f"Invalid expression syntax: {e}") from e

    # 3. Recursively evaluate only whitelisted nodes
    def _eval(node):
        if isinstance(node, ast.Expression):
            return _eval(node.body)
        if isinstance(node, ast.Constant):
            if isinstance(node.value, (int, float)) and not isinstance(node.value, bool):
                return node.value
            raise ValueError("Only numeric constants allowed")
        if isinstance(node, ast.BinOp):
            op = _ALLOWED_OPERATORS.get(type(node.op))
            if op is None:
                raise ValueError(f"Operator not allowed: {type(node.op).__name__}")
            left, right = _eval(node.left), _eval(node.right)
            # DoS guard for exponentiation
            if isinstance(node.op, ast.Pow) and abs(right) > MAX_POWER_EXPONENT:
                raise ValueError("Exponent too large")
            return op(left, right)
        if isinstance(node, ast.UnaryOp):
            op = _ALLOWED_OPERATORS.get(type(node.op))
            if op is None:
                raise ValueError("Unary operator not allowed")
            return op(_eval(node.operand))
        # Anything else (Call, Attribute, Name, Subscript, Import...) is rejected
        raise ValueError(f"Disallowed expression element: {type(node).__name__}")

    return _eval(tree)
```

**Why this works:** `ast.parse` never executes code. The recursive evaluator only handles `Constant`, `BinOp`, and `UnaryOp` — so `Call`, `Attribute`, `Name`, `Import`, and `Subscript` nodes (the attack surface) are structurally impossible to reach.
